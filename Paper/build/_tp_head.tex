\section{The two-path family: outcome trap and matched theorem}
\label{app:two-path}
This appendix proves \cref{thm:matched}. \Cref{app:two-path-trap} constructs the
suboptimal outcome minimum; \cref{app:random-trap-certificate} proves outcome
trapping from zero-readout seeds; \cref{app:matched-proof} proves the process
half and combines the two. Throughout, $K=2$, $0<p<1/2$, $\lambda=1-2p$,
$h=\lambda^{n-1}$, $H=\log((1-p)/p)$, the start is uniform and
$\ell(x)=\log(1+e^{-x})$.

\paragraph{The family.} Recall \cref{eq:two-path-model}: with $S=(-1)^{s_{\mathrm{prev}}}$,
$C=(-1)^{b_{\mathrm{gate}}}$, $u\in[0,1]$ the mean bit value in a supplied averaging mask,
$W=6(n+1)$ units and all parameters $P=(a,g,q,k,v,w,d,b)$ trainable,
\[
 f_P(S,C,u)=S\bigl(a+g\tanh(qk)\,C\bigr)
 +\sum_{i=1}^{W}v_i\relu(w_i u+d_iS+b_i).
\]
At transition $t\in\{2,\ldots,n\}$ the inputs are $s_{t-1}$, $b_t$ and the mean of the first $t$ bits,
and the label is $Y=(-1)^{s_t}=SC$; the step loss averages over $t$. At the answer the
inputs are $s_0$, $b_n$ and the mean of all bits, and the label is $Y=(-1)^{s_n}$.
We call $a,g,v$ the \emph{readouts}, $c=g\tanh(qk)$ the \emph{gate contrast}, and $d$ the
\emph{state weights} of the averaging units.

\begin{proposition}[An attracting suboptimal outcome minimum]
\label{prop:two-path-trap}
The family contains positive-margin executors for both formats. Its population outcome
loss has a non-strict local minimum with logits
\[
 f_{\mathrm{trap}}(S,C,u)=m_{\mathrm{trap}}SC,\qquad m_{\mathrm{trap}}=\log\frac{1+h}{1-h},
 \qquad \Pr(Yf_{\mathrm{trap}}>0)=\frac{1+h}2<1,
\]
and loss $L_{\mathrm{trap}}>0$, although the loss infimum over the family is zero.
An open neighborhood of this minimum flows into it under outcome gradient flow.
\end{proposition}

\begin{proposition}[Outcome trapping from zero readout]
\label{prop:random-outcome-trap}
Let the seed have $a=g=v=0$, have every ReLU active on all of $\{\pm1\}\times[0,1]$ or inactive on
all of it, and satisfy the certificate \cref{eq:outcome-certificate} below. Then outcome gradient flow
stays in that activation region, keeps loss at least $L_{\mathrm{trap}}$, and converges to a local
minimum with logits $f_{\mathrm{trap}}$; its limiting accuracy is $(1+h)/2$. The state weights need not vanish.
\end{proposition}
